توابع Classification
make_blobs

تولید خوشه‌های گاوسی ایزوتروپیک. ساده‌ترین تابع برای ایجاد داده‌های چندکلاسه با خوشه‌های کروی‌شکل. مناسب برای تست K-Means و SVM.

کد نمونه
from sklearn.datasets import make_blobs
import matplotlib.pyplot as plt

# تولید ۳۰۰ نمونه با ۳ خوشه
X, y = make_blobs(
    n_samples=300,
    n_features=2,
    centers=3,
    cluster_std=1.8,
    random_state=42
)

plt.scatter(X[:, 0], X[:, 1], c=y, cmap='viridis')
plt.title('make_blobs: 3 clusters')
plt.show()

make_classification

تولید دیتاست طبقه‌بندی با کنترل دقیق ویژگی‌های اطلاعاتی، افزونه و نویزی. امکان تنظیم درجه سختی مسئله، تعداد کلاس‌ها و وزن کلاس‌ها را دارد.

کد نمونه
from sklearn.datasets import make_classification

X, y = make_classification(
    n_samples=400,
    n_features=2,
    n_informative=2,
    n_redundant=0,
    n_classes=3,
    n_clusters_per_class=1,
    class_sep=1.5,
    random_state=42
)

# شکل داده: X→(400,2) y→(400,)
print(X.shape, y.shape)

make_gaussian_quantiles

تولید داده از یک توزیع گاوسی چندمتغیره و تقسیم آن به کوانتیل‌های برابر. مرزهای تصمیم منحنی‌شکل ایجاد می‌کند که برای تست مدل‌های غیرخطی مناسب است.

کد نمونه
from sklearn.datasets import make_gaussian_quantiles
import matplotlib.pyplot as plt

X, y = make_gaussian_quantiles(
    mean=None,
    cov=1.0,
    n_samples=300,
    n_features=2,
    n_classes=3,
    random_state=42
)

# مرزهای دایره‌ای/بیضوی بین کلاس‌ها
plt.scatter(X[:,0], X[:,1], c=y, cmap='plasma')
نمایش خروجی
make_moons

تولید دو نیم‌دایره درهم‌تنیده (هلال ماه). یک بنچمارک کلاسیک برای آزمایش الگوریتم‌هایی که مرزهای تصمیم غیرخطی را یاد می‌گیرند مثل Kernel SVM.

کد نمونه
from sklearn.datasets import make_moons
import matplotlib.pyplot as plt

X, y = make_moons(
    n_samples=300,
    noise=0.15,
    random_state=42
)

# ۲ کلاس با مرز غیرخطی هلالی
plt.scatter(X[:,0], X[:,1],
           c=y, cmap='coolwarm',
           edgecolors='k', s=30)

make_circles

تولید دو دایره تودرتو — کلاس داخلی و کلاس خارجی. مسئله به‌صورت خطی جداناپذیر است و برای تست Kernel SVM، شبکه‌های عصبی و روش‌های مبتنی بر فاصله ایده‌آل است.

کد نمونه
from sklearn.datasets import make_circles
import matplotlib.pyplot as plt

X, y = make_circles(
    n_samples=300,
    noise=0.08,
    factor=0.5,
    random_state=42
)

# factor: نسبت شعاع دایره داخلی به خارجی
plt.scatter(X[:,0], X[:,1],
           c=y, cmap='bwr',
           edgecolors='k', s=30)

توابع Regression
make_regression

تولید دیتاست رگرسیون خطی با کنترل تعداد ویژگی‌های اطلاعاتی (n_informative)، میزان نویز، بایاس و تعداد خروجی‌ها. پرکاربردترین تابع برای تست مدل‌های رگرسیونی.

کد نمونه
from sklearn.datasets import make_regression
import matplotlib.pyplot as plt

X, y = make_regression(
    n_samples=200,
    n_features=1,
    n_informative=1,
    noise=20.0,
    random_state=42
)

# مشاهده رابطه خطی با نویز
plt.scatter(X, y, alpha=0.6, s=20)
plt.xlabel('Feature')
plt.ylabel('Target')
نمایش خروجی
make_friedman1

بنچمارک کلاسیک Friedman #1 با رابطه غیرخطی: y = 10·sin(π·x₁x₂) + 20(x₃−0.5)² + 10x₄ + 5x₅ + noise. فقط ۵ ویژگی اول اثر دارند و بقیه نویز خالص‌اند؛ مناسب تست مدل‌های قدرتمند مثل GBM.

کد نمونه
from sklearn.datasets import make_friedman1

X, y = make_friedman1(
    n_samples=300,
    n_features=10,
    noise=1.0,
    random_state=42
)

# فقط ۵ ویژگی اول اطلاعاتی‌اند
# بقیه نویز خالص‌اند
print(f"X: {X.shape}, y: {y.shape}")
# X: (300, 10), y: (300,)
نمایش خروجی — توزیع y
make_friedman2

بنچمارک Friedman #2: y = √(x₁² + (x₂x₃ − 1/(x₂x₄))²) + noise. تعداد ویژگی‌ها ثابت و برابر ۴ است و مقیاس آن‌ها بسیار متفاوت است؛ برای سنجش حساسیت مدل به مقیاس ویژگی‌ها مفید است.

کد نمونه
from sklearn.datasets import make_friedman2

X, y = make_friedman2(
    n_samples=300,
    noise=0.0,
    random_state=42
)

# ۴ ویژگی با بازه‌های متفاوت:
# x1∈[0,100], x2∈[40π,560π]
# x3∈[0,1],   x4∈[1,11]
print(X.min(axis=0).round(1))
# [  0.5 133.2   0.    1.1]
نمایش خروجی — توزیع y
make_friedman3

بنچمارک Friedman #3: y = arctan((x₂x₃ − 1/(x₂x₄)) / x₁) + noise. برخلاف friedman2، خروجی در بازه محدود (π/2, π/2-) قرار می‌گیرد؛ مناسب مدل‌هایی با خروجی کران‌دار.

کد نمونه
from sklearn.datasets import make_friedman3

X, y = make_friedman3(
    n_samples=300,
    noise=0.0,
    random_state=42
)

# y ∈ (−π/2, π/2) — بازه محدود
print(f"y range: [{y.min():.3f}, {y.max():.3f}]")
# y range: [0.200, 1.570]
نمایش خروجی — توزیع y
make_sparse_uncorrelated

تولید دیتاست رگرسیون با ۴ ویژگی اطلاعاتی و ویژگی‌های اضافه‌ی ناهمبسته. مناسب برای ارزیابی روش‌های Feature Selection مثل Lasso و ElasticNet.

کد نمونه
from sklearn.datasets import make_sparse_uncorrelated

X, y = make_sparse_uncorrelated(
    n_samples=100,
    n_features=10,
    random_state=42
)

# y = x1 + 2·x2 − 2·x3 − 1.5·x4
# بقیه ویژگی‌ها هیچ اثری روی y ندارند
print(X.shape, y.shape)
# (100, 10) (100,)
